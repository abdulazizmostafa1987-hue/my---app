
name: Build Android APK

on:
  workflow_dispatch:
  push:
    branches: [main, master]
    paths:
      - 'baqa_hesba.zip'
      - '.github/workflows/build.yml'

permissions:
  contents: read

concurrency:
  group: baqa-hesba-android-${{ github.ref }}
  cancel-in-progress: true

jobs:
  build:
    name: Analyze, test and build APK
    runs-on: ubuntu-24.04
    timeout-minutes: 60

    env:
      KEYSTORE_PASSWORD: ${{ secrets.KEYSTORE_PASSWORD }}
      KEY_PASSWORD: ${{ secrets.KEY_PASSWORD }}
      KEY_ALIAS: ${{ secrets.KEY_ALIAS }}

    steps:
      - name: Checkout repository
        uses: actions/checkout@v6

      - name: Find and extract Flutter project
        id: project
        shell: bash
        run: |
          set -euo pipefail

          if [[ -f "$GITHUB_WORKSPACE/baqa_hesba.zip" ]]; then
            SOURCE_ROOT="$RUNNER_TEMP/baqa_source"
            mkdir -p "$SOURCE_ROOT"
            unzip -oq "$GITHUB_WORKSPACE/baqa_hesba.zip" -d "$SOURCE_ROOT"
          else
            SOURCE_ROOT="$GITHUB_WORKSPACE"
          fi

          PUBSPEC="$(find "$SOURCE_ROOT" -type f -name pubspec.yaml \
            -not -path '*/.dart_tool/*' \
            -not -path '*/build/*' -print -quit)"

          if [[ -z "$PUBSPEC" ]]; then
            echo 'ERROR: pubspec.yaml was not found.' >&2
            exit 1
          fi

          PROJECT_DIR="$(dirname "$PUBSPEC")"

          [[ -f "$PROJECT_DIR/lib/main.dart" ]] || {
            echo "ERROR: lib/main.dart missing" >&2
            exit 1
          }

          [[ -f "$PROJECT_DIR/ci/prepare_android.sh" ]] || {
            echo 'ERROR: ci/prepare_android.sh is missing.' >&2
            exit 1
          }

          echo "project_dir=$PROJECT_DIR" >> "$GITHUB_OUTPUT"
          echo "Flutter project: $PROJECT_DIR"

      - name: Set up Java 17
        uses: actions/setup-java@v6
        with:
          distribution: temurin
          java-version: '17'

      - name: Install Flutter stable without cache
        shell: bash
        run: |
          set -euo pipefail

          FLUTTER_DIR="$RUNNER_TEMP/flutter-sdk"

          git clone --depth 1 --branch stable \
            https://github.com/flutter/flutter.git \
            "$FLUTTER_DIR"

          echo "$FLUTTER_DIR/bin" >> "$GITHUB_PATH"
          export PATH="$FLUTTER_DIR/bin:$PATH"

          flutter --version
          flutter config --no-analytics
          flutter precache --android

      - name: Configure Android SDK
        shell: bash
        run: |
          set -euo pipefail

          SDK_ROOT="${ANDROID_SDK_ROOT:-${ANDROID_HOME:-/usr/local/lib/android/sdk}}"

          if [[ ! -d "$SDK_ROOT" ]]; then
            echo "ERROR: Android SDK not found: $SDK_ROOT" >&2
            exit 1
          fi

          SDKMANAGER="$(find "$SDK_ROOT/cmdline-tools" -type f \
            -name sdkmanager -print -quit 2>/dev/null || true)"

          if [[ -z "$SDKMANAGER" ]]; then
            echo 'ERROR: sdkmanager not found.' >&2
            exit 1
          fi

          echo "ANDROID_HOME=$SDK_ROOT" >> "$GITHUB_ENV"
          echo "ANDROID_SDK_ROOT=$SDK_ROOT" >> "$GITHUB_ENV"
          echo "$(dirname "$SDKMANAGER")" >> "$GITHUB_PATH"
          echo "$SDK_ROOT/platform-tools" >> "$GITHUB_PATH"

          yes | "$SDKMANAGER" --licenses >/dev/null 2>&1 || true

          "$SDKMANAGER" --install \
            'platform-tools' \
            'platforms;android-36' \
            'build-tools;36.0.0' \
            'ndk;27.0.12077973'

          test -f "$SDK_ROOT/ndk/27.0.12077973/source.properties"

      - name: Install SQLite native headers
        run: |
          sudo apt-get update
          sudo apt-get install -y libsqlite3-dev

      - name: Configure optional release keystore
        env:
          KEYSTORE_BASE64: ${{ secrets.KEYSTORE_BASE64 }}
        shell: bash
        run: |
          set -euo pipefail

          if [[ -n "${KEYSTORE_BASE64:-}" ]]; then
            : "${KEYSTORE_PASSWORD:?Missing KEYSTORE_PASSWORD}"
            : "${KEY_PASSWORD:?Missing KEY_PASSWORD}"
            : "${KEY_ALIAS:?Missing KEY_ALIAS}"

            KEYSTORE_FILE="$RUNNER_TEMP/baqa-release.jks"
            printf '%s' "$KEYSTORE_BASE64" | base64 --decode > "$KEYSTORE_FILE"

            test -s "$KEYSTORE_FILE"
            echo "KEYSTORE_PATH=$KEYSTORE_FILE" >> "$GITHUB_ENV"
          else
            echo 'No private keystore; Flutter default signing will be used.'
          fi

      - name: Prepare Android project
        working-directory: ${{ steps.project.outputs.project_dir }}
        run: bash ci/prepare_android.sh

      - name: Resolve dependencies
        working-directory: ${{ steps.project.outputs.project_dir }}
        run: flutter pub get

      - name: Analyze Dart code
        working-directory: ${{ steps.project.outputs.project_dir }}
        run: flutter analyze --no-pub

      - name: Run automated tests
        working-directory: ${{ steps.project.outputs.project_dir }}
        run: flutter test --no-pub

      - name: Build release APK
        working-directory: ${{ steps.project.outputs.project_dir }}
        run: flutter build apk --release --no-pub

      - name: Verify APK and signature
        id: apk
        working-directory: ${{ steps.project.outputs.project_dir }}
        shell: bash
        run: |
          set -euo pipefail

          APK_PATH='build/app/outputs/flutter-apk/app-release.apk'
          test -s "$APK_PATH"

          APKSIGNER="$(find "$ANDROID_SDK_ROOT/build-tools" \
            -type f -name apksigner -print | sort -V | tail -1)"

          [[ -n "$APKSIGNER" ]] || {
            echo 'ERROR: apksigner not found.' >&2
            exit 1
          }

          "$APKSIGNER" verify "$APK_PATH"

          echo "apk_path=$PWD/$APK_PATH" >> "$GITHUB_OUTPUT"
          sha256sum "$APK_PATH"
          ls -lh "$APK_PATH"

      - name: Upload release APK
        uses: actions/upload-artifact@v6
        with:
          name: baqa-hesba-release-apk
          path: ${{ steps.apk.outputs.apk_path }}
          if-no-files-found: error
          retention-days: 14
